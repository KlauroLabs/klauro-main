import Client, { Injectable as Service, LIMITS, RemoteError } from '@scope/client';
import { localHelper } from './local-helper';

/** Module-level documentation. */
export const moduleValue: number = 3;

@Service({ retries: 2, labels: ['primary', 'fallback'] })
export abstract class BaseRunner implements Runnable {
  @Service('token')
  public readonly name?: string = 'runner';

  constructor(private readonly client: Client) {}

  /** Execute one request. */
  async run(input: string = 'default'): Promise<string> {
    if (LIMITS.enabled && input) {
      for (const item of [input]) {
        await this.client.send(item);
      }
    } else {
      throw new RemoteError('disabled');
    }

    return Promise.resolve(input).then((value) => localHelper(value));
  }

  protected abstract stop(): void;
}

export const helpers = {
  format(value: string) {
    try {
      return localHelper(value ?? 'none');
    } catch (error) {
      throw RemoteError(error);
    }
  },
};

export function choose(flag: boolean): number {
  const nested = () => {
    if (flag) return LIMITS.maximum;
    throw new RemoteError('nested');
  };
  return flag ? nested() : 0;
}
